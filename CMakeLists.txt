cmake_minimum_required(VERSION 3.20)
project(MintPlayer VERSION 0.1 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Musxi Player requires Windows 10 or newer")
endif()
option(MUSXI_ENABLE_CEF "Build the isolated CEF SDK compatibility targets" OFF)
# Keep the CRT identical across the C++17/20 static-library boundary (CEF uses /MT).
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
function(musxi_standard target standard)
  set_target_properties(${target} PROPERTIES CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  endif()
endfunction()
add_library(music_core STATIC src/player/player_service.cpp)
musxi_standard(music_core 17)
add_library(music_application STATIC src/main.cpp)
musxi_standard(music_application 17)
target_compile_definitions(music_application PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(music_application PRIVATE music_core gdiplus gdi32 winmm comdlg32 shell32 ole32 dwmapi crypt32)
add_executable(MintPlayer WIN32 src/application/legacy_entry.cpp src/version.rc)
set_target_properties(MintPlayer PROPERTIES OUTPUT_NAME "MusxiPlayer")
musxi_standard(MintPlayer 17)
target_compile_definitions(MintPlayer PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(MintPlayer PRIVATE music_application)
if(MINGW)
  target_link_options(MintPlayer PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
# SDK configuration must remain in this child scope; never include it above.
if(MUSXI_ENABLE_CEF)
  add_subdirectory(src/cef)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(test_name smoke player-interface frame-smoke)
    add_executable(${test_name} tests/${test_name}.cpp)
    musxi_standard(${test_name} 17)
    target_link_libraries(${test_name} PRIVATE music_core gdiplus gdi32 winmm comdlg32 shell32 ole32 dwmapi crypt32)
    if(MINGW)
      target_link_options(${test_name} PRIVATE -static)
      if(NOT test_name STREQUAL "player-interface")
        target_link_options(${test_name} PRIVATE -municode)
      endif()
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  add_executable(standard-boundary tests/standard-boundary.cpp)
  musxi_standard(standard-boundary 20)
  target_link_libraries(standard-boundary PRIVATE music_application)
  if(MINGW)
    target_link_options(standard-boundary PRIVATE -static)
  endif()
  add_test(NAME standard-boundary COMMAND standard-boundary)
  if(TARGET cef_host)
    add_test(NAME cef-ipc-smoke
      COMMAND "$<TARGET_FILE_DIR:cef_host>/MusxiPlayerWeb.exe" --cef-smoke)
    set_tests_properties(cef-ipc-smoke PROPERTIES TIMEOUT 35 RUN_SERIAL TRUE)
  endif()
endif()

cmake_minimum_required(VERSION 3.20)
project(MintPlayer VERSION 0.1 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Mint Player requires Windows 10 or newer")
endif()
add_executable(MintPlayer WIN32 src/main.cpp src/version.rc)
set_target_properties(MintPlayer PROPERTIES OUTPUT_NAME "MusxiPlayer")
target_compile_features(MintPlayer PRIVATE cxx_std_17)
target_compile_definitions(MintPlayer PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(MintPlayer PRIVATE gdiplus gdi32 winmm comdlg32 shell32 ole32 dwmapi crypt32)
if(MINGW)
  target_link_options(MintPlayer PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(MintPlayer PRIVATE /utf-8 /W4)
endif()

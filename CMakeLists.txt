cmake_minimum_required(VERSION 3.20)
project(MusxiPlayer VERSION 0.3.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Musxi Player requires Windows 10 or newer")
endif()
option(MUSXI_ENABLE_CEF "Build the CEF/Vue application" ON)
option(MUSXI_ENABLE_FFMPEG "Build the FFmpeg/WASAPI backend" ON)
if(MUSXI_ENABLE_CEF AND NOT MUSXI_ENABLE_FFMPEG)
  message(FATAL_ERROR "The CEF application requires FFmpeg/WASAPI; use both options OFF for a Core-only build.")
endif()
option(MUSXI_TEST_AUDIO_DEVICE "Run audible A3 WASAPI checks at 10 percent stream volume" OFF)
# Keep the CRT identical across the C++17/20 static-library boundary (CEF uses /MT).
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
function(musxi_standard target standard)
  set_target_properties(${target} PROPERTIES CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  endif()
endfunction()
add_subdirectory(src/audio)
add_library(music_core STATIC src/player/player_service.cpp)
musxi_standard(music_core 17)
if(MUSXI_ENABLE_FFMPEG)
  add_library(music_application STATIC src/application/application.cpp)
  musxi_standard(music_application 17)
  target_compile_definitions(music_application PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(music_application PRIVATE music_core music_audio_wasapi shell32 ole32 crypt32)
endif()

  add_subdirectory(src/cef)
